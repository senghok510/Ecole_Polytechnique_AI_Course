
-- We will work with a table triples1, 11.000.000 integer triples;
-- 0<=s, o <= 8000000;
-- 0 <= p < 20

\timing
-- to know how long each operation takes

create table triples1(s int, p int, o int);
\d triples1;
-- load the table: 
copy triples1 from '/Users/ioana.manolescu/GITLAB/inf553/trunk/2025/SLIDES/5-OPERATORS/triples1.csv';
-- this took  3635,060 ms
-- 3635/11000000 = .00033045454545454545 ms per tuple, that's not a lot!

-- create an index: 
-- B+ tree index created on s; unclustered

create index on triples1(s);
\d triples1; 

-- let's see if the index is used! Explain asks Postgres to tell us
-- how it would evaluate the query (not evaluate it) 
explain select o from triples1 where s < 2900000; 
-- with more details:  
explain (verbose) select o from triples1 where s < 2900000 ;
-- explain analyze also runs the query, thus we get two sets of
-- information:
-- - the optimizer's estimations of the execution times and numbers
--    of results
-- - the actual execution time and number of results
explain (analyze, costs, verbose) select o from triples1 where s < 2900000 ;

-- we can change the constant to be basically as high as the largest
-- value; in this case, the query returns all tuples; 
explain (analyze, costs, verbose) select o from triples1 where s < 8000000 ;

-- vacuum optimizes the disk usage (compacts the data pages in files)
-- analyze is asks PostgreSQL to build more detailed statistics about the table and
-- store them. After analyze, PostgreSQL has a better idea about the
-- properties of the data and may better optimize a query
vacuum analyze; 

-- inspect the plans Postgres makes now: 
explain select o from triples1 where s < 8000000;

explain select o from triples1 where s < 2000000;

explain select o from triples1 where s < 4000000;

-- cluster the index: 11 million records are physically recopied
-- in the stable storage, to put them in the leaves of the index!
cluster triples1 using triples1_s_idx;
-- This took 19507,594 ms (5 times more than loading the table)!
-- B+tree bulkloading is used
-- Still, because the index is clustered, few records fit in a data
-- leaf page
\d triples1;

-- inspect more plans: 
explain select o from triples1 where s < 2900000; -- selective
explain select o from triples1 where s < 8000000; -- not selective at all

-- run again vacuum analyze to make sure the optimizer has
-- the latest information about our table: 
vacuum analyze; 

-- re-inspect the plans: 
explain select o from triples1 where s < 2900000;
explain select o from triples1 where s < 8000000;

-- insert a tuple in the table, with the lowest s value: 
-- (at the lowest left leaf in the index):
insert into triples1 values (0, 0, 0);
-- this took 1,958 ms, basically 2ms for a single tuple!
-- contrast with the high efficiency for bulk loading the relation
-- (above) as long as there was no index
-- now insert a tuple with highest s value so far: 
insert into triples values (9000000, 100, 100);
-- this took 1,150 ms, still high

-- let's redo things in another order: 
drop tables triples1; 
-- this also drops the index
-- re-create the table: 
create table triples1(s int, p int, o int);
-- create the index and cluster it now (on an empty table) 
create index on triples1(s);
cluster triples1 using triples1_s_idx;
-- now reload the table:
copy triples1 from '/Users/ioana.manolescu/GITLAB/inf553/trunk/2025/SLIDES/5-OPERATORS/triples1.csv';
-- This took 25788,998 ms instead of 3635 ms without the index!
-- Again, we pay the cost of node splits.

-- if we make the selection more "fuzzy" at the end of the string, but
-- not at the beginning, the index is still used! Note the rewriting
-- of the selection predicate (the system has transformed the query
-- into an interval one): 
imdb=# explain analyze select title, production_year from title where title like 'Godzilla%'; 
                                                         QUERY PLAN                                                         
----------------------------------------------------------------------------------------------------------------------------

 Planning Time: 0.122 ms
 Execution Time: 0.207 ms
